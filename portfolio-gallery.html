AI / LLM PII Redaction Logic

Hybrid rule-based + LLM verification pass that scrubs PII before content ships (Project 1).

Redaction pass 
# pii_redaction.py — hybrid rule + LLM pass
import re
from langchain.chat_models import ChatOpenAI

RULES = [
    (re.compile(r"\b\d{10}\b"), "[PHONE]"),
    (re.compile(r"[\w.+-]+@[\w-]+\.[\w.-]+"), "[EMAIL]"),
]
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

def rule_pass(text: str) -> str:
    for pattern, tag in RULES:
        text = pattern.sub(tag, text)
    return text

def llm_verify(text: str) -> str:
    # catches names, addresses, context-only PII the rules miss
    prompt = f"Redact remaining PII in:\n{text}"
    return llm.invoke(prompt).content

def redact(raw: str) -> str:
    return llm_verify(rule_pass(raw))  # same input → same output

Web Scraping & Sentiment Scoring

Scraping and scoring sentiment across 200+ websites on an automated schedule (Project 5).

Scrape & score pipeline 
# sentiment_pipeline.py
from bs4 import BeautifulSoup
import requests

def fetch_text(url: str) -> str:
    soup = BeautifulSoup(
        requests.get(url, timeout=10).content, "html.parser"
    )
    return " ".join(p.get_text() for p in soup.find_all("p"))

def score_sentiment(text: str) -> float:
    pos = {"good", "strong", "growth"}
    neg = {"bad",  "weak",   "decline"}
    w   = text.lower().split()
    s   = sum(x in pos for x in w) - sum(x in neg for x in w)
    return s / max(len(w), 1)
